# Para automatizar a iniciação do backend criar um arquivo inicializar.sh com o seguinte comando
#! /bin/bash

pid_py=$(ps aux | grep '[m]eu-projeto/venv/bin/python main.py' | awk '{print $2}')

if [ -z $pid_py ]
then
        pid_py=-1
fi

echo "Achado os seguintes pid:"
echo $pid_py

if [ $pid_py -ne -1 ]
then
        echo "O Servidor Python do Meu Projeto esta em execucao"
else
        path_pwd=$(pwd)
        echo "O Servidor Python do Meu Projeto nao estava em execucao"
        cd $path_pwd
        # Caso queira criar o venv
        # $path_pwd/venv/bin/python -m venv venv
        # Ativar o venv já existente
        source  ./venv/bin/activate
        ./venv/bin/pip install -r requirements.txt
        nohup $path_pwd/venv/bin/python main.py 2>output.out &
fi
